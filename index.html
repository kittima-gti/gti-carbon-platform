1.6 Anaerobic Pond — บ่อบำบัดน้ำเสียแบบไม่เติมอากาศ
CH₄(kg) = factor × [(Wi × CODin/1000) - S] | TGO ฉบับที่ 7
≤2m: 0.05>2m: 0.20▼
ℹ️
สูตร TGO ฉบับที่ 7: CH₄(kg) = factor × [(Wi × CODin/1000) − S] | Wi=ปริมาณน้ำเสีย (m³), CODin=COD ขาเข้า (mg/L), S=COD ที่ถูกกำจัดเป็นตะกอน (kg, ปกติ≈0 ถ้าไม่นำตะกอนออก) | factor: ≤2m=0.05, ไม่ดูแล=0.075, >2m=0.20 | CF = CH₄ × GWP28

พบปัญหา "Script function not found: doGet"? — สาเหตุ: วางโค้ดใน Apps Script ไม่ครบ หรือ Deploy ก่อนวางโค้ด | วิธีแก้: (1) ลบโค้ดเดิมออกทั้งหมด (2) วางโค้ดใหม่จากด้านล่างนี้ทั้งหมด รวม doPost และ doGet (3) กด Deploy → New deployment อีกครั้ง (4) เลือก URL ใหม่มาวาง | โค้ดต้องมีทั้ง function doPost(e) และ function doGet(e)
🔵
เห็น ActivityLog ขึ้นแต่ RawData/Summary ไม่ขึ้น + Console error 400?
แปลว่าโค้ด Apps Script เวอร์ชันเก่ายัง error ตอน parse | วิธีแก้ (ทำตามลำดับ):
1. ลบ Sheet ActivityLog เดิมที่ค้างออก (ถ้ามี)
2. เปิด Apps Script → ลบโค้ดเก่าทั้งหมด → วางโค้ดใหม่ด้านล่างนี้ (เวอร์ชัน 4.1 robust)
3. กด Deploy → Manage deployments → ✏️ Edit → Version: New version → Deploy (ใช้ URL เดิมได้)
4. กลับมากด "🧪 ทดสอบ" → ดูว่ามีชีต RawData + แถว TEST ปรากฏ
💡 โค้ดใหม่จะสร้างชีต Errors อัตโนมัติถ้ามี error — เปิดดูเพื่อ debug ได้
⚠️
เช็ค 3 จุดสำคัญเมื่อข้อมูลไม่เข้า Sheet:
1. URL ต้องลงท้ายด้วย /exec (ไม่ใช่ /dev)
2. Deploy → ตั้ง "Who has access" = Anyone (สำคัญที่สุด! ถ้าตั้งเป็น "Only myself" ข้อมูลจะไม่เข้า)
3. หลังแก้โค้ดทุกครั้ง ต้อง Deploy ใหม่ หรือ Manage deployments → Edit (ดินสอ) → Version: New version มิฉะนั้นโค้ดเก่ายังทำงานอยู่

// ═══ Google Apps Script — GTI Carbon Platform v4 (Robust) ═══ // วางโค้ดนี้ทั้งหมดใน Apps Script Editor แล้ว Deploy ใหม่

function doPost(e) {
  try {
    // อ่าน payload — รองรับทั้ง postData.contents และ parameter
    var raw = (e && e.postData && e.postData.contents) ? e.postData.contents : '{}';
    var data = JSON.parse(raw);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var ts = new Date();
    var rows = data.rows || [];
    // ─── RawData ───
    var rawSh = ss.getSheetByName('RawData');
    if (!rawSh) rawSh = ss.insertSheet('RawData');
    if (rawSh.getLastRow() === 0) rawSh.appendRow(['Timestamp','Period','Facility','User','Scope','Category','Activity','Qty','Unit','EF','kgCO2e','tCO2e','EF_Source']);
    for (var i=0; i<rows.length; i++) {
      var r = rows[i];
      rawSh.appendRow([ts, data.period||'', data.facility||'', data.user||'', r.scope||'', r.cat||'', r.act||'', r.qty||0, r.unit||'', r.ef||0, r.kgco2e||0, r.tco2e||0, r.src||'']);
    }
    // ─── Summary ───
    var sumSh = ss.getSheetByName('Summary');
    if (!sumSh) sumSh = ss.insertSheet('Summary');
    if (sumSh.getLastRow() === 0) sumSh.appendRow(['Timestamp','Period','Facility','User','S1_tCO2e','S2_tCO2e','S3_tCO2e','Total_tCO2e']);
    sumSh.appendRow([ts, data.period||'', data.facility||'', data.user||'', data.s1||0, data.s2||0, data.s3||0, data.total||0]);
    // ─── ActivityLog ───
    if (data.log && data.log.length) {
      var logSh = ss.getSheetByName('ActivityLog');
      if (!logSh) logSh = ss.insertSheet('ActivityLog');
      if (logSh.getLastRow() === 0) logSh.appendRow(['Timestamp','User','Role','Action','Detail']);
      for (var j=0; j<data.log.length; j++) {
        var l = data.log[j];
        logSh.appendRow([l.ts||'', l.user||'', l.role||'', l.action||'', l.detail||'']);
      }
    }
    return ContentService.createTextOutput(JSON.stringify({status:'ok', rows:rows.length})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    // บันทึก error ลงชีต Errors เพื่อ debug
    try {
      var es = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Errors') || SpreadsheetApp.getActiveSpreadsheet().insertSheet('Errors');
      es.appendRow([new Date(), err.toString(), (e && e.postData ? e.postData.contents : 'no postData')]);
    } catch(e2) {}
    return ContentService.createTextOutput(JSON.stringify({status:'error', msg:err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({status:'GTI Carbon Platform v4 OK', version:'4.1'})).setMimeType(ContentService.MimeType.JSON);
}
